# 优化项：自动发现新类别

## 当前状态

意图类别固定为 7 类（+ general），新增类别完全靠人工。用户问了很多"口腔问题"、"新生儿护理"等问题，但因为没建类别，全部掉进 general。

## 改进方向

定期（每周/攒够 N 条）将 general 中的"漏网之鱼"发给 LLM 做聚类分析，自动发现新主题，人工审核后加入规则库。

---

## 方案

```
1. 收集阶段：把路由结果为 general 的问题攒下来
2. 聚类阶段：攒够 N 条后发给 LLM，让它找主题
3. 审核阶段：人工确认后，一键加入 KEYWORD_RULES + INTENT_APPENDS
```

---

## 代码

```python
"""
新意图发现 — 从 general 问题中自动聚类，发现未覆盖的主题

使用方式：
    # 1. 收集（在线，在 chat_service.py 中加一行日志）
    python -m docs.optimization-plans.intent_discovery collect

    # 2. 聚类分析（离线，定期跑）
    python -m docs.optimization-plans.intent_discovery analyze

    # 3. 查看结果
    python -m docs.optimization-plans.intent_discovery report
"""

import json
import os
from datetime import datetime
from collections import defaultdict
from typing import Optional

from app.services.llm import call_deepseek
from app.models.chat import ChatMessage, ChatOptions


# -----------------------------------------------------------
# 配置
# -----------------------------------------------------------

# general 问题的存储路径
DATA_DIR = os.path.join(os.path.dirname(__file__), "..", "data")
GENERAL_QUESTIONS_FILE = os.path.join(DATA_DIR, "general_questions.jsonl")

# 最少出现次数 — 低于此值的主题不值得建类别
MIN_CLUSTER_SIZE = 5

# 分析时取最近多少条 general 问题
MAX_QUESTIONS_FOR_ANALYSIS = 200


# -----------------------------------------------------------
# 1. 收集：在线记录 general 问题
# -----------------------------------------------------------

def collect(question: str, date: Optional[str] = None) -> None:
    """
    将一条被分类为 general 的问题写入日志文件。

    调用位置：route_intent() 中，LLM 分类结果确认为 general 后调用。
    
    集成方式（在 intent_router.py 中）：
        if llm_intent == "general":
            from docs.optimization_plans.intent_discovery import collect
            collect(question)
    """
    os.makedirs(DATA_DIR, exist_ok=True)

    record = {
        "date": date or datetime.now().strftime("%Y-%m-%d"),
        "question": question,
    }

    with open(GENERAL_QUESTIONS_FILE, "a", encoding="utf-8") as f:
        f.write(json.dumps(record, ensure_ascii=False) + "\n")


# -----------------------------------------------------------
# 2. 分析：定期跑 LLM 聚类
# -----------------------------------------------------------

def load_recent_questions(limit: int = MAX_QUESTIONS_FOR_ANALYSIS) -> list[str]:
    """加载最近 N 条 general 问题"""
    if not os.path.exists(GENERAL_QUESTIONS_FILE):
        return []

    questions = []
    with open(GENERAL_QUESTIONS_FILE, "r", encoding="utf-8") as f:
        for line in f:
            try:
                record = json.loads(line)
                questions.append(record["question"])
            except json.JSONDecodeError:
                pass

    # 取最近 N 条
    return questions[-limit:]


def analyze() -> list[dict]:
    """
    将最近 general 问题发给 LLM 做聚类分析。

    Returns:
        LLM 发现的新意图列表，每个意图包含：
          - intent: 英文 slug（如 "newborn_care", "oral_health"）
          - label: 中文标签（如 "新生儿护理", "口腔健康"）
          - keywords: 15-20 个关键词
          - append_prompt: 该领域的回答侧重点
    """
    questions = load_recent_questions()

    if len(questions) < MIN_CLUSTER_SIZE:
        print(f"general 问题数量不足（{len(questions)} < {MIN_CLUSTER_SIZE}），跳过分析")
        return []

    print(f"加载了 {len(questions)} 条 general 问题，开始 LLM 聚类分析...")

    question_list = "\n".join([f"- {q}" for q in questions])

    discovery_prompt = f"""你是一个数据分析助手。以下是用户向育儿助手提问但被分类为"其他（general）"的问题列表（共 {len(questions)} 条）：

{question_list}

请分析这些问题的主题分布，找出出现次数 >= {MIN_CLUSTER_SIZE} 次且不属于以下已有类别的新主题：
已有类别：illness(疾病) / feeding(喂养) / sleep(睡眠) / development(发育) / emergency(急救) / daily_care(日常护理) / vaccine(疫苗)

为每个达到阈值的新主题按以下 JSON 格式输出。如果没有达到阈值的新主题，返回空数组 []。

[{{ "intent": "newborn_care", "label": "新生儿护理", "keywords": ["黄疸", "脐带", "胎便", "新生儿", "月子", "满月", "剃胎毛", "马牙", "螳螂嘴", "生理性黄疸", "母乳性黄疸", "肚脐渗血", "胎毒", "新生儿痤疮", "惊跳反射"], "append_prompt": "## 👶 新生儿护理模式\\\\n- 强调0-28天新生儿与婴儿的区别（器官未成熟、免疫弱、变化快）\\\\n- 区分生理现象和病理问题（如生理性黄疸 vs 病理性黄疸）\\\\n- 就医指征更保守，新生儿病情进展快" }}]

要求：
1. keywords 选 15-20 个，应该包括正式医学术语 + 用户口语化表达
2. append_prompt 1-2 句话，描述该领域回答时的特殊要求
3. 只返回 JSON，不要加解释文字"""

    messages = [ChatMessage(role="user", content=discovery_prompt)]
    options = ChatOptions(temperature=0.3, maxTokens=2000)

    try:
        raw = call_deepseek(messages, options)
        # 提取 JSON 部分（去掉 LLM 可能在前后加的说明文字）
        start = raw.find("[")
        end = raw.rfind("]") + 1
        if start != -1 and end > start:
            raw = raw[start:end]
        return json.loads(raw)
    except Exception as e:
        print(f"LLM 聚类分析失败: {e}")
        return []


# -----------------------------------------------------------
# 3. 报告：展示分析结果
# -----------------------------------------------------------

def report() -> None:
    """展示分析结果，方便人工审核"""

    print("=" * 60)
    print("新意图发现 — 分析报告")
    print(f"数据来源: {GENERAL_QUESTIONS_FILE}")
    print(f"分析时间: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    print("=" * 60)

    results = analyze()

    if not results:
        print("\n未发现达到阈值的新主题。")
        print("建议：继续收集 general 问题，下周再跑一次。")
        return

    print(f"\n发现 {len(results)} 个新主题：\n")

    for i, item in enumerate(results, 1):
        print(f"{'─' * 40}")
        print(f"【{i}】{item['label']} (intent: {item['intent']})")
        print(f"{'─' * 40}")
        print(f"  关键词 ({len(item['keywords'])}个): "
              f"{', '.join(item['keywords'][:10])}...")
        print(f"  回答侧重: {item['append_prompt'][:100]}...")

    print(f"\n{'─' * 40}")
    print("下一步操作：")
    print("1. 人工审核以上新主题，确认值得建类别")
    print("2. 在 intent_router.py 中：")
    for item in results:
        print(f"   - Intent 类型加 '{item['intent']}'")
        print(f"   - KEYWORD_RULES 加 KeywordRule(keywords=[...], "
              f"intent='{item['intent']}', confidence=0.9)")
        print(f"   - INTENT_APPENDS['{item['intent']}'] = \"\"\"...\"\"\"")
        print(f"   - VALID_INTENTS 加 '{item['intent']}'")


# -----------------------------------------------------------
# 4. 辅助：统计当前 general 分布
# -----------------------------------------------------------

def stats() -> dict:
    """统计 general 问题的数量和时间分布"""
    if not os.path.exists(GENERAL_QUESTIONS_FILE):
        return {"total": 0, "by_date": {}}

    total = 0
    by_date = defaultdict(int)

    with open(GENERAL_QUESTIONS_FILE, "r", encoding="utf-8") as f:
        for line in f:
            try:
                record = json.loads(line)
                total += 1
                by_date[record.get("date", "unknown")] += 1
            except json.JSONDecodeError:
                pass

    return {
        "total": total,
        "by_date": dict(sorted(by_date.items())),
    }


# -----------------------------------------------------------
# CLI 入口
# -----------------------------------------------------------

if __name__ == "__main__":
    import sys

    if len(sys.argv) < 2:
        print("用法:")
        print("  python intent_discovery.py stats    — 查看收集统计")
        print("  python intent_discovery.py report   — 完整分析报告")
        sys.exit(0)

    cmd = sys.argv[1]

    if cmd == "stats":
        s = stats()
        print(f"已收集 {s['total']} 条 general 问题")
        print("按日期分布:", json.dumps(s["by_date"], ensure_ascii=False, indent=2))

    elif cmd == "report":
        report()

    else:
        print(f"未知命令: {cmd}")
        print("可用: stats, report")
